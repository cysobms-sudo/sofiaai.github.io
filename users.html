⏳ در حال دریافت اطلاعات...
'; fetch('/api/users') .then(response => response.json()) .then(data => { if (data.length === 0) { container.innerHTML = '

'; return; } let html = ` `; data.forEach((user, index) => { // ===== تبدیل تاریخ میلادی به شمسی در مرورگر ===== const date = new Date(user.created_at); const persianDate = new Intl.DateTimeFormat('fa-IR', { year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', hour12: false }).format(date); html += ` `; }); html += `
# نام و نام خانوادگی ایمیل شماره تماس تاریخ ثبت‌نام
${index + 1} ${user.name} ${user.email} ${user.phone || '-'} ${persianDate}
🔹 تعداد کل کاربران: ${data.length}
`; container.innerHTML = html; }) .catch(error => { console.error('❌ خطا:', error); container.innerHTML = '
❌ خطا در دریافت اطلاعات. لطفاً دوباره تلاش کنید.
'; }); }
